{{- /* docs/html-attrs Input: dict of attributes Output: string like: key="value" key2="value2" */ -}} {{- $attrs := . | default (dict) -}} {{- $pairs := slice -}} {{- range $k, $v := $attrs -}} {{- if and $v (ne (printf "%v" $v) "") -}} {{- $pairs = $pairs | append (printf `%s="%v"` $k $v) -}} {{- end -}} {{- end -}} {{- return (delimit $pairs " ") -}}
